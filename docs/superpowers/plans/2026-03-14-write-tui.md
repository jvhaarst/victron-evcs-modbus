```python
def validate_write_value(reg: Register | StringRegister, input_str: str) -> int | str:
    """Parse and validate a human-readable display value for writing to reg.

    For Register: parses as float, multiplies by scale, rounds to int, validates
    type bounds. Returns raw int ready for write_register.
    For StringRegister: validates UTF-8 byte length. Returns input_str unchanged.
    Raises ValueError with a descriptive message on invalid input.
    """
    if isinstance(reg, StringRegister):
        encoded = input_str.encode("utf-8")
        max_bytes = reg.register_count * 2
        if len(encoded) > max_bytes:
            raise ValueError(
                f"String too long: {len(encoded)} bytes, max {max_bytes}"
            )
        return input_str
    try:
        display_val = float(input_str)
    except ValueError:
        raise ValueError(f"Invalid number: {input_str!r}")
    raw = round(display_val * reg.scale)
    if reg.data_type == "uint16":
        if not (0 <= raw <= 65535):
            raise ValueError(
                f"Value out of range for uint16 (0–65535 raw, got {raw})"
            )
    else:  # int16
        if not (-32768 <= raw <= 32767):
            raise ValueError(
                f"Value out of range for int16 (-32768–32767 raw, got {raw})"
            )
    return raw


def write_register(
    client: Any,
    reg: Register | StringRegister,
    value: int | str,
) -> bool:
    """Write a single register via Modbus FC6 (scalar) or FC16 (string).

    value must be a validated raw int (Register) or str (StringRegister).
    For int16 registers, value is a signed Python int (e.g. -500); wire
    encoding (adding 65536) is handled internally.
    Returns True on success, False on any failure (Modbus error or exception).
    """
    try:
        if isinstance(reg, StringRegister):
            raw_bytes = value.encode(reg.encoding)
            padded = raw_bytes.ljust(reg.register_count * 2, b"\x00")
            words = list(struct.unpack(f">{reg.register_count}H", padded))
            response = client.write_registers(
                address=reg.address, values=words, device_id=1
            )
        else:
            wire_value = (value + 65536) if value < 0 else value
            response = client.write_register(
                address=reg.address, value=wire_value, device_id=1
            )
        return not response.isError()
    except Exception:
        return False
```

from __future__ import annotations

import struct

from textual.app import App, ComposeResult
from textual.binding import Binding
from textual.containers import Horizontal, Vertical

from textual.widgets import Button, DataTable, Footer, Header, Input, Label

class EditModal(ModalScreen[int | str | None]):
    """Modal screen for editing a single register value."""

    BINDINGS = [Binding("escape", "cancel", "Cancel", show=False)]

    DEFAULT_CSS = """
    EditModal {
        align: center middle;
    }
    #edit-dialog {
        background: $panel;
        border: tall $primary;
        padding: 1 2;
        width: 60;
        height: auto;
    }
    #edit-hint {
        margin-top: 1;
        color: $text-muted;
    }
    """

    def __init__(self, reg: Register | StringRegister, current_display: str) -> None:
        super().__init__()
        self.reg = reg
        self.current_display = current_display

    def compose(self) -> ComposeResult:
        unit = getattr(self.reg, "unit", "")
        unit_str = f"  [{unit}]" if unit else ""
        with Vertical(id="edit-dialog"):
            yield Label(f"[bold]{self.reg.name}[/bold]{unit_str}")
            yield Label(f"Address: {self.reg.address}")
            yield Input(value=self.current_display, id="edit-input")
            yield Label("Enter to confirm  •  Esc to cancel", id="edit-hint")

    def on_mount(self) -> None:
        self.query_one("#edit-input", Input).focus()

    def on_input_submitted(self, event: Input.Submitted) -> None:
        try:
            raw = validate_write_value(self.reg, event.value)
            self.dismiss(raw)
        except ValueError as e:
            self.query_one("#edit-hint", Label).update(f"[red]{e}[/red]")

    def action_cancel(self) -> None:
        self.dismiss(None)


class ConfirmModal(ModalScreen[str]):
    """Generic confirmation modal with configurable buttons."""

    DEFAULT_CSS = """
    ConfirmModal {
        align: center middle;

    """Full-screen Textual app for browsing and editing Victron EVCS registers."""

    BINDINGS = [
        Binding("ctrl+s", "write_all", "Write", show=True),

    DataTable {

    }

        registers: list[Register | StringRegister],
        values: dict[int, int | str | None],
    ) -> None:
        super().__init__()
        self.modbus_client = client
        self.registers = registers
        self.values: dict[int, int | str | None] = dict(values)

        table = self.query_one(DataTable)
        table.clear(columns=True)
        table.add_columns("Address", "Name", "Value", "Unit", "Default", "Access")
        self._reg_by_key.clear()
        for reg in self.registers:
            key = str(reg.address)
            self._reg_by_key[key] = reg
            table.add_row(
                str(reg.address),
                reg.name,
                self._value_cell(reg),
                getattr(reg, "unit", ""),
                self._default_cell(reg),
                reg.access,
                key=key,
            )
        self._update_subtitle()

    def _value_cell(self, reg: Register | StringRegister) -> str:
        address = reg.address
        raw = self.pending.get(address, self.values.get(address))
        marker = "* " if address in self.pending else ""
        if raw is None:
            return f"{marker}[red]error[/red]"
        if isinstance(reg, StringRegister):
            return f"{marker}{raw}"
        return f"{marker}{format_value(raw, reg.scale)}"

    def _default_cell(self, reg: Register | StringRegister) -> str:
        if isinstance(reg, StringRegister):
            return ""
        return format_value(reg.default, reg.scale) if reg.default is not None else ""

    def _refresh_row(self, address: int) -> None:
        reg = self._reg_by_key.get(str(address))
        if reg is None:
            return
        table = self.query_one(DataTable)
        table.update_cell(str(address), "Value", self._value_cell(reg), update_width=True)

    def _update_subtitle(self) -> None:
        count = len(self.pending)
        self.sub_title = f"{count} pending" if count else ""

    def on_data_table_row_selected(self, event: DataTable.RowSelected) -> None:
        key = event.row_key.value
        reg = self._reg_by_key.get(key)
        if reg is None or reg.access != "R/W":
            return
        raw = self.pending.get(reg.address, self.values.get(reg.address))
        if raw is None:
            current_display = ""
        elif isinstance(reg, StringRegister):
            current_display = str(raw)
        else:
            current_display = format_value(raw, reg.scale)

        def handle_edit(result: int | str | None) -> None:
            if result is not None:
                self.pending[reg.address] = result
                self._refresh_row(reg.address)
                self._update_subtitle()

        self.push_screen(EditModal(reg, current_display), handle_edit)

    def action_write_all(self) -> None:
        if not self.pending:
            self.notify("Nothing to write", severity="information")
            return
        failed_address = None
        written: list[int] = []
        for address, raw_value in list(self.pending.items()):
            reg = self._reg_by_key.get(str(address))
            if reg is None:
                continue
            if write_register(self.modbus_client, reg, raw_value):
                self.values[address] = raw_value
                del self.pending[address]  # remove only successfully written
                written.append(address)
                self._refresh_row(address)
            else:
                failed_address = address
                break  # stop; failed register + all subsequent remain in self.pending
        self._update_subtitle()
        if failed_address is not None:
            self.notify(
                f"Write failed at address {failed_address} — check connection",
                severity="error",
            )
        elif written:
            self.notify(f"Written {len(written)} register(s)", severity="information")

    def action_refresh_registers(self) -> None:
        if not self.pending:
            self._do_refresh()
            return

        def handle_confirm(choice: str) -> None:
            if choice == "Yes":
                self.pending.clear()
                self._do_refresh()

        self.push_screen(
            ConfirmModal(
                f"Refreshing will discard {len(self.pending)} unsaved change(s). Continue?",
                ["Yes", "No"],
            ),
            handle_confirm,
        )

    def _do_refresh(self) -> None:
        try:
            new_values = read_registers(self.modbus_client, self.registers)
            self.values = new_values
            self._rebuild_table()
            self.notify("Refreshed", severity="information")
        except Exception as e:
            self.notify(f"Refresh failed: {e}", severity="error")

    def action_quit_app(self) -> None:
        if not self.pending:
            self.exit()
            return

        def handle_confirm(choice: str) -> None:
            if choice == "Yes":
                self.action_write_all()
                self.exit()
            elif choice == "No":
                self.exit()
            # Cancel: do nothing (modal already dismissed)

        self.push_screen(
            ConfirmModal(
                f"You have {len(self.pending)} unsaved change(s). Write before quitting?",
                ["Yes", "No", "Cancel"],
            ),
            handle_confirm,
        )
